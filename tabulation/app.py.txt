from flask import Flask, request, render_template
import csv
import os

app = Flask(__name__)

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/submit_score', methods=['POST'])
def save_score():
    # 1. Grab all the specific criteria from the form
    candidate = request.form['candidate_number']
    production = int(request.form['production'])
    swimwear = int(request.form['swimwear'])
    formalwear = int(request.form['formalwear'])
    qa = int(request.form['qa'])
    
    # 2. Automatically calculate the total score
    total_score = production + swimwear + formalwear + qa
    
    # 3. Save all individual scores plus the total into the CSV
    with open('pageant_scores.csv', mode='a', newline='') as file:
        writer = csv.writer(file)
        writer.writerow(["Candidate " + candidate, production, swimwear, formalwear, qa, total_score])
        
    return "Score submitted successfully!"

if __name__ == '__main__':
    # Updated headers to match the new criteria
    if not os.path.exists('pageant_scores.csv'):
        with open('pageant_scores.csv', mode='w', newline='') as file:
            writer = csv.writer(file)
            writer.writerow(['Candidate', 'Production (20)', 'Swimwear (25)', 'Formal Wear (25)', 'Q&A (30)', 'Total Score'])

    print("Starting server... Connect judges to your IPv4 address!")
    app.run(host='0.0.0.0', port=80)
